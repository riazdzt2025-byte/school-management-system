from django.test import TestCase
from django.urls import reverse

class MissingRollLab(TestCase):

        self.student = Student.objects.create(name='Lab Student', admission_class='9',

from django.test import Client

# students/management/commands/report_missing_contact.py
from django.core.management.base import BaseCommand
from students.models import Student

class Command(BaseCommand):
    help = "অভিভাবকের যোগাযোগ নম্বর ছাড়া সক্রিয় শিক্ষার্থীর সংখ্যা দেখায়"

    def add_arguments(self, parser):
        parser.add_argument('--class-name', default=None)

    def handle(self, *args, **opts):
        qs = Student.objects.filter(guardian_contact_no='', is_archived=False)
        if opts['class_name']:
            qs = qs.filter(admission_class=opts['class_name'])
        self.stdout.write(f"missing contact: {qs.count()}")
        for s in qs[:20]:
            self.stdout.write(f"  {s.student_id} {s.name} {s.admission_class}")
